29| function loadSettings() {
30|   try {
31|     const j = JSON.parse(localStorage.getItem(STORE) || "{}");
32|     Object.assign(MENU.S, j);
33|     if (typeof MENU.S.camX !== "number") MENU.S.camX = null;
34|   } catch { /* first run */ }
35| }
36| function saveSettings() {
37|   try { localStorage.setItem(STORE, JSON.stringify(MENU.S)); } catch { /* private mode */ }
38| }

40| function segRow(container, values, selected, cb) {
41|   const r = document.createElement("div");
42|   r.className = "menu-seg";
43|   for (const v of values) {
44|     const b = document.createElement("button");
45|     b.textContent = v.label;
46|     b.dataset.v = v.id;
47|     if (v.id === selected) b.classList.add("on");
48|     b.addEventListener("click", () => {
49|       for (const x of r.children) x.classList.remove("on");
50|       b.classList.add("on");
51|       beep(360, 460, 0.03, "sine", 0.02);
52|       cb(v.id);
53|       syncSummary();
54|     });
55|     r.appendChild(b);
56|   }
57|   container.appendChild(r);
58|   return r;
59| }

61| function mapThumb(cv, mapDef) {
62|   // Miniature preview rendered from the same seeded map — cheap (tiny tiles).
63|   const m = generateMap({ w: mapDef.w, h: mapDef.h, seed: mapDef.seed, scatter: 0.18, edges: 5 });
64|   const TW = Math.max(1, Math.round(mapDef.w / 45)), TH = Math.max(1, Math.round(mapDef.h / 45));
65|   const s = generateTileSheet(m.tiles, TW, TH);
66|   const ctx = cv.getContext("2d");
67|   ctx.imageSmoothingEnabled = false;
68|   ctx.clearRect(0, 0, cv.width, cv.height);
69|   ctx.drawImage(s, m.sx, m.sy, (TW * 2 + 2), (TH * 2 + 2), 0, 0, cv.width, cv.height);
70| }

72| function toggleRow(parent, label, get, set) {
73|   const row = document.createElement("div");
74|   row.className = "menu-row";
75|   const lab = document.createElement("span");
76|   lab.textContent = label;
77|   row.appendChild(lab);
78|   const y = document.createElement("div");
79|   y.className = "menu-y menu-seg";
80|   for (const val of [true, false]) {
81|     const b = document.createElement("button");
82|     b.textContent = val ? "ON" : "OFF";
83|     if (get() === val) b.classList.add("on");
84|     b.addEventListener("click", () => {
85|       for (const x of y.children) x.classList.remove("on");
86|       b.classList.add("on");
87|       set(val);
88|       beep(360, 440, 0.03, "sine", 0.02);
89|     });
90|     y.appendChild(b);
91|   }
92|   row.appendChild(y);
93|   parent.appendChild(row);
94| }

96| const summary = () => document.querySelector("#sk-summary");
97| function syncSummary() {
98|   const el = summary();
99|   if (!el) return;
100|   const f = document.querySelector("#sk-fac .on")?.dataset.v || "china";
101|   const ai = document.querySelector("#sk-ai .on")?.dataset.v || "japan";
102|   const r = document.querySelector("#sk-res .on")?.dataset.v || "medium";
103|   const d = document.querySelector("#sk-diff .on")?.dataset.v || "medium";
104|   const map = document.querySelector("#sk-maps .on")?.dataset.v || MAPS[0].id;
105|   const m = MAPS.find((x) => x.id === map) || MAPS[0];
106|   el.textContent =
107|     `${FACTION_META[f].nameEN} vs ${FACTION_META[ai].nameEN} — ${m.name} — ` +
108|     `funds ${r.toUpperCase()} — AI ${AI_DIFF[d].label}`;
109| }

149| function show(i) {
150|   const scr = (id) => document.getElementById(id);
151|   scr("ms-title").hidden = i !== 0;
152|   if (i === 1) buildSkirmish();
153|   if (i === 2) buildSettings();
154|   scr("ms-skirmish").hidden = i !== 1;
155|   scr("ms-settings").hidden = i !== 2;
156|   if (i === 1) { syncSummary(); markAiWarning(); }
157|   if (i === 2) refreshCamLabel();
158|   window.scrollTo(0, 0);
159| }

161| function onMenuKey(e) {
162|   if (started) return;
163|   if (e.key === "Escape" &&
164|       (!document.getElementById("ms-skirmish").hidden || !document.getElementById("ms-settings").hidden)) {
165|     beep(500, 320, 0.05, "triangle");
166|     show(0);
167|   }
168| }

318|     MENU.S.musicMuted = !v; sfx && (sfx.musicMuted = MENU.S.musicMuted); saveSettings();
319|   });
320|   toggleRow(document.getElementById("st-audio"), "SFX volume", () => MENU.S.sfxVol > 0, (v) => {
321|     MENU.S.sfxVol = v ? 1 : 0.25; applySfx(); saveSettings();
322|   });
323| 
324|   toggleRow(document.getElementById("st-game"), "Screen shake", () => MENU.S.shake, (v) => {
325|     MENU.S.shake = v; TUNE.shake = v; saveSettings();
326|   });
327|   toggleRow(document.getElementById("st-game"), "Show grid (debug)", () => MENU.S.showGrid, (v) => {
328|     MENU.S.showGrid = v; saveSettings();
329|   });
330| 
331|   document.getElementById("st-back").addEventListener("click", () => { beep(500, 320, 0.05, "triangle"); show(0); });
332|   document.getElementById("st-close").addEventListener("click", () => { beep(500, 322, 0.05, "triangle"); show(0); });
333|   refreshCamLabel();
334| }

336| function refreshCamLabel() {
337|   const el = document.getElementById("st-cam");
338|   if (!el) return;
339|   el.textContent = MENU.S.camX != null
340|     ? `saved at tile (${Math.floor(MENU.S.camX / 32)}, ${Math.floor((MENU.S.camY || 0) / 32)})`
341|     : "not yet saved";
342| }

344| // main.js calls this once the sound system exists (or after settings edits)
345| export function setSfx(ref) {
346|   sfx = ref;
347|   applySfx();
348| }
349| function applySfx() {
350|   if (!sfx) return;
351|   if (sfx.applySettings) sfx.applySettings(MENU.S);
352| }

357|   MENU.S.camY = y;
